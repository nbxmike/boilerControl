cmake_minimum_required(VERSION 3.20.0)
project(router_host_tests LANGUAGES C)
enable_testing()
add_executable(router_tests test_router.c ../src/router.c)
target_include_directories(router_tests PRIVATE ../include)
target_compile_features(router_tests PRIVATE c_std_11)
if(MSVC)
  target_compile_options(router_tests PRIVATE /W4 /WX /UNDEBUG)
else()
  target_compile_options(router_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME router_behavior COMMAND router_tests)
