cmake_minimum_required(VERSION 3.20)
project(valve_host_tests LANGUAGES CXX)
enable_testing()

foreach(model BTS7960 VICTOR_SPX TALON_SRX JAGUAR_GREY VICTOR_888 VICTOR_884)
  add_executable(valve_model_${model} test_valve_control.cpp)
  target_include_directories(valve_model_${model} PRIVATE ../include)
  target_compile_features(valve_model_${model} PRIVATE cxx_std_17)
  target_compile_definitions(valve_model_${model} PRIVATE
    MOTOR_CONTROLLER=MOTOR_CONTROLLER_${model})
  if(MSVC)
    target_compile_options(valve_model_${model} PRIVATE /W4 /WX /UNDEBUG)
  else()
    target_compile_options(valve_model_${model} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  endif()
  add_test(NAME valve_model_${model} COMMAND valve_model_${model})
endforeach()

add_executable(valve_half_speed test_valve_control.cpp)
target_include_directories(valve_half_speed PRIVATE ../include)
target_compile_features(valve_half_speed PRIVATE cxx_std_17)
target_compile_definitions(valve_half_speed PRIVATE MOTOR_CONTROLLER=MOTOR_CONTROLLER_VICTOR_SPX
  FRC_PWM_SPEED_PERMILLE=500 FRC_PWM_PERIOD_US=20000)
if(MSVC)
  target_compile_options(valve_half_speed PRIVATE /W4 /WX /UNDEBUG)
else()
  target_compile_options(valve_half_speed PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME valve_half_speed COMMAND valve_half_speed)
